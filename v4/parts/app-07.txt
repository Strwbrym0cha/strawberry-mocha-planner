function handleForm(form){const Date().toISOString(),completedAt:'',source:'manual'});persist('Sweet To-Do added')}else if(type==='edit-task'){state.life.tasks=state.life.tasks.map(t=>t.id===d.id?{...t,text:d.text,date:d.date,priority:d.priority,minutes:Number(d.minutes)||15,energy:d.energy,location:d.location,protected:bool(d.protected)}:t);dialog=null;persist('Task updated')}else if(type==='ping'){state.life.reminders.push({id:makeId('ping'),title:d.title,date:d.date,time:d.time,timing:d.timing,completed:false,createdAt:new Date().toISOString(),source:'manual'});persist('Little Ping added')}else if(type==='edit-ping'){state.life.reminders=state.life.reminders.map(r=>r.id===d.id?{...r,title:d.title,date:d.date,time:d.time}:r);dialog=null;persist('Little Ping updated')}else if(type==='event'){state.life.events.push({id:makeId('event'),title:d.title,type:d.type,date:d.date,startTime:d.startTime,endTime:d.endTime,location:d.location,createdAt:new Date().toISOString()});persist('Berry Busy item added')}else if(type==='mochini'){const result=processMochini(state,d.message);state=result.state;state=saveState(state);render()}else if(type==='routine'){const steps=text(d.steps).split('\n').map((label,i)=>({id:makeId(`step${i}`),label:text(label),minutes:5,optional:false})).filter(x=>x.label);state.life.routines.push({id:makeId('routine'),name:d.name,icon:d.icon||'🎀',daypart:d.daypart,recurrence:d.recurrence,days:[],steps,archived:false,createdAt:new Date().toISOString()});persist('Routine added')}else if(type==='food'){state.nourish.noms.foods.push({id:makeId('food'),name:d.name,kind:'other',tags:[],effort:'low',prepMinutes:0,available:Number(d.quantity)>0,trackQuantity:true,quantity:Math.max(0,Number(d.quantity)||0),lowAt:Math.max(0,Number(d.lowAt)||1),createdAt:new Date().toISOString()});persist('Stocked Nom added')}else if(type==='grocery'){state.nourish.noms.groceries=[...list(state.nourish.noms.groceries),{id:makeId('grocery'),name:d.name,quantity:d.quantity,checked:false,createdAt:new Date().toISOString()}];persist('Added to grocery basket')}else if(type==='meal'){state.nourish.noms.mealPlan=[...list(state.nourish.noms.mealPlan),{id:makeId('meal'),label:d.label,date:d.date,slot:d.slot,sourceType:'manual',createdAt:new Date().toISOString()}];persist('Nom planned')}else if(type==='recipe'){state.nourish.noms.recipes=[...list(state.nourish.noms.recipes),{id:makeId('recipe'),name:d.name,prepMinutes:Number(d.prepMinutes)||15,ingredients:text(d.ingredients).split(',').map(text).filter(Boolean),available:true,createdAt:new Date().toISOString()}];persist('Recipe saved')}else if(type==='nom-log'){state.nourish.noms.history.push({id:makeId('nom-log'),name:d.name,sourceType:'manual',sourceId:'',kind:'other',date:today(),loggedAt:new Date().toISOString()});persist('Nom logged')}else if(type==='sip'){state.nourish.sips.waterGoalOz=Math.max(8,Number(d.goal)||64);state.nourish.sips.history.push({id:makeId('sip'),name:d.name||d.type,ounces:Number(d.ounces)||0,type:d.type,date:today(),loggedAt:new Date().toISOString(),source:'manual'});persist('Sip logged')}else if(type==='motion'){state.movement.sessions.push({id:makeId('motion'),type:d.type,label:d.type[0].toUpperCase()+d.type.slice(1),minutes:Number(d.minutes)||0,effort:d.effort,date:d.date,loggedAt:new Date().toISOString(),source:'manual'});persist('Movement logged')}else if(type==='person'){state.v4.people.push({id:makeId('person'),name:d.name,relationship:d.relationship,likes:d.likes,importantDates:d.importantDates,giftIdeas:d.giftIdeas,createdAt:new Date().toISOString()});persist('Person added')}else if(type==='hobby'){state.v4.hobbies.push({id:makeId('hobby'),name:d.name,status:d.status,kind:d.kind,language:bool(d.language),notes:d.notes,lastTouched:'',createdAt:new Date().toISOString()});persist('Hobby added')}else if(type==='work-item'){state.work.items.push({id:makeId('work-item'),text:d.text,dueDate:d.dueDate,minutes:Number(d.minutes)||15,done:false,protected:bool(d.protected),createdAt:new Date().toISOString()});persist('Work item added')}else if(type==='shift'){state.work.shifts.push({id:makeId('shift'),label:d.label||'Work shift',date:d.date,startTime:d.startTime,endTime:d.endTime,status:'planned',createdAt:new Date().toISOString()});persist('Shift added')}else if(type==='bill'){state.money.bills.push({id:makeId('bill'),name:d.name,amount:Number(d.amount)||0,dueDate:d.dueDate,dueDay:Number(d.dueDay)||0,recurring:true,paid:false,createdAt:new Date().toISOString()});persist('Bill added')}else if(type==='debt'){state=upsertDebt(state,{type:d.type,name:d.name,balance:d.balance,apr:d.apr,paymentAmount:d.paymentAmount,minimum:d.paymentAmount,dueDay:d.dueDay,creditLimit:d.creditLimit,lender:d.lender});persist('Debt added and linked bill created')}else if(type==='edit-debt'){state=upsertDebt(state,{id:d.id,type:d.type,name:d.name,balance:d.balance,apr:d.apr,paymentAmount:d.paymentAmount,minimum:d.paymentAmount,dueDay:d.dueDay,creditLimit:d.creditLimit});dialog=null;persist('Debt and linked bill updated')}else if(type==='earning'){state.money.earnings.push({id:makeId('earning'),kind:'paycheck',label:d.label,status:d.status,expectedDate:d.date,receivedDate:d.status==='received'?d.date:'',estimatedGross:Number(d.amount)||0,receivedAmount:d.status==='received'?Number(d.amount)||0:0,createdAt:new Date().toISOString()});persist('Income added')}else if(type==='savings'){state.money.savingsGoals.push({id:makeId('savings'),name:d.name,current:Number(d.current)||0,target:Number(d.target)||0,createdAt:new Date().toISOString()});persist('Savings goal added')}else if(type==='spending'){state.money.spending.push({id:makeId('spend'),description:d.description,amount:Number(d.amount)||0,category:d.category,date:d.date,loggedAt:new Date().toISOString(),source:'manual'});persist('Spending logged')}else if(type==='wish'){state.v4.shopping.push({id:makeId('wish'),name:d.name,price:Number(d.price)||0,category:d.category,why:d.why,waitUntil:d.waitUntil,bought:false,createdAt:new Date().toISOString()});persist('Added to wishlist')}else if(type==='course'){state.education.courses.push({id:makeId('course'),title:d.title,progress:Number(d.progress)||0,currentObjective:d.currentObjective,nextMilestone:d.nextMilestone,targetDate:d.targetDate,status:'active',createdAt:new Date().toISOString()});persist('Course added')}else if(type==='study-item'){state.education.items.push({id:makeId('study-item'),courseId:d.courseId,title:d.title,type:'study',dueDate:d.dueDate,minutes:Number(d.minutes)||15,energy:'medium',initiation:'easy',difficulty:'medium',protected:false,done:false,createdAt:new Date().toISOString()});persist('Study item added')}else if(type==='study-session'){state.education.sessions.push({id:makeId('study-session'),courseId:d.courseId,date:d.date,minutes:Number(d.minutes)||0,completedAt:new Date().toISOString()});persist('Study session logged')}else if(type==='thread'){state.life.threads.push({id:makeId('thread'),title:d.title,icon:'🧵',summary:d.summary,status:d.status,deadline:d.deadline,notes:[],resources:[],links:{},createdAt:new Date().toISOString(),updatedAt:new Date().toISOString()});persist('Thread started')}else if(type==='goal'){state.growth.goals.push({id:makeId('goal'),title:d.title,why:d.why,targetDate:d.targetDate,status:'moving',kind:'outcome',milestones:[],createdAt:new Date().toISOString()});persist('Goal added')}else if(type==='win'){state.growth.wins.push({id:makeId('win'),title:d.title,note:d.note,date:today(),source:'manual',createdAt:new Date().toISOString()});persist('Win saved')}else if(type==='dump'){state.v4.brainDump.push({id:makeId('dump'),text:d.text,bucket:d.bucket,createdAt:new Date().toISOString()});persist(d.bucket==='closed'?'Locked in Closed Drawer':'Brain dump saved')}else if(type==='admin'){state.v4.admin.push({id:makeId('admin'),name:d.name,kind:d.kind,physicalLocation:d.physicalLocation,digitalLocation:d.digitalLocation,expires:d.expires,notes:d.notes,createdAt:new Date().toISOString()});persist('Life-admin record saved')}else if(type==='review'){const existing=list(state.insights.dayReviews).find(x=>x.date===today()),r={...existing,id:existing?.id||makeId('review'),date:today(),mood:d.mood,energy:d.energy,happened:d.happened,proud:d.proud,tomorrow:d.tomorrow,helped:d.helped,hard:d.hard,snapshot:reviewSnapshot(today()),createdAt:existing?.createdAt||new Date().toISOString(),updatedAt:new Date().toISOString()};state.insights.dayReviews=[...list(state.insights.dayReviews).filter(x=>x.date!==today()),r];persist('Day in Review saved')}else if(type==='blocks'){const blocks=text(d.blocks).split('\n').map((line,i)=>{const [name,start,end,energy]=line.split('|').map(text);return{id:makeId(`block${i}`),name:name||`Block ${i+1}`,start,end,energy:['low','okay','high','unknown'].includes(energy)?energy:'okay'}}).filter(x=>x.start&&x.end);state.v4.energyBlocks=blocks;dialog=null;persist('Energy blocks updated')}}

function approveProposal(){const p=state.mochini?.pendingProposal;if(!p)return;if(p.kind==='task'){state.life.tasks.push({id:makeId('task'),...p.payload,priority:p.payload.date===today()?'today':'whenever',done:false,createdAt:new Date().toISOString(),source:'mochini'})}else if(p.kind==='reminder'){state.life.reminders.push({id:makeId('ping'),...p.payload,completed:false,createdAt:new Date().toISOString(),source:'mochini'})}else if(p.kind==='task-batch'){for(const item of p.payload.items)state.life.tasks.push({id:makeId('task'),...item,priority:item.date===today()?'today':'whenever',done:false,createdAt:new Date().toISOString(),source:'mochini'})}state.mochini.pendingProposal=null;state.mochini.conversation.push({id:makeId('turn'),role:'assistant',text:'Done 😊 It’s in the right room now.',at:new Date().toISOString()});persist('Mochini action approved')}

function deleteByKind(kind,id){if(kind==='task')state.life.tasks=state.life.tasks.filter(x=>x.id!==id);else if(kind==='reminder')state.life.reminders=state.life.reminders.filter(x=>x.id!==id);else if(kind==='debt'){state=removeDebt(state,id)}else if(kind==='person')state.v4.people=state.v4.people.filter(x=>x.id!==id);else if(kind==='nom-log')state.nourish.noms.history=state.nourish.noms.history.filter(x=>x.id!==id);else if(kind==='sip')state.nourish.sips.history=state.nourish.sips.history.filter(x=>x.id!==id);else if(kind==='motion-session')state.movement.sessions=state.movement.sessions.filter(x=>x.id!==id);else if(kind==='grocery')state.nourish.noms.groceries=state.nourish.noms.groceries.filter(x=>x.id!==id);persist('Deleted')}
