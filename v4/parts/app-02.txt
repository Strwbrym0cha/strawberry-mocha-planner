function routineOccurs(r,date=today()){if(r.archived||archived('routine',r.id))return false;const d=new Date(`${date}T12:00:00`),day=d.getDay();if(r.recurrence==='manual')return !!list(state.life.routineInstances).find(x=>x.routineId===r.id&&x.date===date);if(r.recurrence==='weekdays')return day>=1&&day<=5;if(r.recurrence==='weekends')return day===0||day===6;if(r.recurrence==='selected')return list(r.days).map(Number).includes(day);return true}
